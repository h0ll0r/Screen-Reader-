 ШАГ 1: Установить Python (если нет)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Скачать: https://www.python.org/downloads/
⚠️ При установке поставь галочку "Add Python to PATH"

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 ШАГ 2: Установить Tesseract OCR
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Скачать: https://github.com/UB-Mannheim/tesseract/wiki
  → Выбери: tesseract-ocr-w64-setup-*.exe

При установке обязательно отметь языки:
  ✅ Russian
  ✅ English (обычно уже выбран)

Стандартный путь установки:
  C:\Program Files\Tesseract-OCR\tesseract.exe

Если установил в другое место — измени TESSERACT_PATH
в файле screen_reader.py (строка 34)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 ШАГ 3: Установить Python-библиотеки
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Открой командную строку (cmd) и выполни:

  pip install mss Pillow pytesseract pyttsx3 keyboard

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 ШАГ 4: Запуск
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
⚠️ ВАЖНО: Запускать от имени администратора!
(библиотека keyboard требует права администратора)

Способ 1 — через cmd (от администратора):
  python screen_reader.py

Способ 2 — правой кнопкой на screen_reader.py
  → "Запустить от имени администратора" (нужен .pyw или ярлык)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 КАК ПОЛЬЗОВАТЬСЯ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. Запусти программу
2. Нажми F9 (или свою клавишу)
3. Откроется скриншот — выдели зону с текстом мышью
4. Нажми Enter
5. Текст будет озвучен!

Esc — отменить выделение
Ctrl+C в консоли — выйти из программы

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 НАСТРОЙКИ (в начале файла screen_reader.py)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
HOTKEY = "f9"          → поменяй на любую клавишу (f1-f12, home, etc.)
TESSERACT_PATH = ...   → путь к tesseract.exe
OCR_LANG = "rus+eng"   → языки распознавания

 ВОЗМОЖНЫЕ ПРОБЛЕМЫ

❌ "tesseract is not installed"
   → Установи Tesseract OCR (Шаг 2) и проверь TESSERACT_PATH

❌ "Access denied" при запуске
   → Запускай cmd от имени администратора

❌ Русский текст не читается
   → При установке Tesseract убедись что выбрал Russian language pack
